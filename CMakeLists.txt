cmake_minimum_required(VERSION 3.16)
project(crawltalk LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CRAWLTALK_GUI "Build the SDL desktop" ON)
option(CRAWLTALK_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(CRAWLTALK_HTTP "Build the libcurl HTTP service" ON)
option(CRAWLTALK_THREAD_SANITIZE "Enable ThreadSanitizer (separate build)" OFF)
if(CRAWLTALK_THREAD_SANITIZE AND CRAWLTALK_SANITIZE)
  message(FATAL_ERROR "ThreadSanitizer and AddressSanitizer require separate builds")
endif()
if(CRAWLTALK_THREAD_SANITIZE)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()
if(CRAWLTALK_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
add_library(crawltalk_vm src/interpreter.cpp src/objmemory.cpp src/bitblt.cpp src/evaluation.cpp src/hostservices.cpp src/hostbridge.cpp)
target_include_directories(crawltalk_vm PUBLIC src)
target_link_libraries(crawltalk_vm PUBLIC Threads::Threads)
if(CRAWLTALK_HTTP)
  find_package(CURL 7.32 REQUIRED)
  target_compile_definitions(crawltalk_vm PRIVATE CRAWLTALK_CURL)
  target_link_libraries(crawltalk_vm PUBLIC CURL::libcurl)
endif()
add_executable(crawltalk-headless src/headless.cpp)
target_link_libraries(crawltalk-headless PRIVATE crawltalk_vm)
if(CRAWLTALK_GUI)
  find_package(SDL2 CONFIG REQUIRED)
  add_executable(Smalltalk src/main.cpp)
  target_compile_definitions(Smalltalk PRIVATE CRAWLTALK_SDL_FLAT)
  target_link_libraries(Smalltalk PRIVATE crawltalk_vm SDL2::SDL2)
endif()
enable_testing()
add_executable(hostservices-test tests/hostservices_test.cpp)
target_link_libraries(hostservices-test PRIVATE crawltalk_vm)
add_test(NAME host_services COMMAND hostservices-test)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_executable(snapshot-test tests/snapshot_test.cpp)
target_link_libraries(snapshot-test PRIVATE crawltalk_vm)
add_test(NAME snapshot_write COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/snapshot_test.py $<TARGET_FILE:snapshot-test>)
add_test(NAME image_foundation COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/image_test.py $<TARGET_FILE:crawltalk-headless>)
if(CRAWLTALK_HTTP)
  add_test(NAME http_service COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/tests/http_test.py $<TARGET_FILE:crawltalk-headless>)
  set_tests_properties(http_service PROPERTIES TIMEOUT 180)
endif()
set_tests_properties(host_services image_foundation PROPERTIES TIMEOUT 180)
